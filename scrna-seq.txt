library(SoupX)
base_dir <- "./00_matrix"
sample_dirs <- list.dirs(base_dir, full.names = TRUE, recursive = FALSE)
tod_list <- list()
toc_list <- list()
for (sample_dir in sample_dirs) {
  sample_name <- basename(sample_dir)
  message("Processing: ", sample_name)
  if (!file.exists(file.path(sample_dir, "matrix.mtx"))) {
    message("  -> matrix.mtx not found, skipping.")
    next
  }
  
  tod <- tryCatch({
    Read10X(data.dir = sample_dir)
  }, error = function(e) {
    message("  -> Failed to read 10X matrix: ", e$message)
    return(NULL)
  })
  
  if (is.null(tod)) next
  
  counts_file <- file.path(sample_dir, paste0(sample_name, "_EmptyDrops_CR_counts.txt"))
  if (!file.exists(counts_file)) {
    message("  -> counts.txt not found: ", counts_file)
    next
  }
  
  counts_info <- read.table(counts_file, header = TRUE, sep = "\t")
  cb_barcodes <- counts_info$Barcode[counts_info$mark == "CB"]
  if (any(grepl("-1$", colnames(tod)))) {
    cb_barcodes <- paste0(cb_barcodes, "-1")
  }
  
  toc <- tod[, colnames(tod) %in% cb_barcodes]
  tod_list[[sample_name]] <- tod
  toc_list[[sample_name]] <- toc
  message("  -> Done: ", sample_name, " | tod: ", ncol(tod), " | toc: ", ncol(toc))
}
sc_list <- mapply(function(sample, tod, toc) {
  message("Creating SoupChannel for: ", sample)
  sc <- SoupChannel(tod, toc)
  return(sc)
}, sample = names(tod_list), tod = tod_list, toc = toc_list, SIMPLIFY = FALSE)
names(sc_list) <- names(tod_list)


library(Seurat)
library(SoupX)
library(DropletUtils)

output_dir <- "./01_soupX_filtered"
dir.create(output_dir, showWarnings = FALSE)

for (sample in names(sc_list)) {
  message("Processing SoupX for sample: ", sample)
  sc <- sc_list[[sample]]
  toc <- sc$toc
  tod <- sc$tod
  seu <- CreateSeuratObject(counts = toc)
  seu <- NormalizeData(seu)
  seu <- FindVariableFeatures(seu, selection.method = "vst", nfeatures = 3000)
  seu <- ScaleData(seu)
  seu <- RunPCA(seu, verbose = FALSE)
  seu <- FindNeighbors(seu, dims = 1:30)
  seu <- FindClusters(seu, resolution = 0.6)
  cluster_info <- setNames(seu$seurat_clusters, colnames(seu))
  sc <- setClusters(sc, clusters = cluster_info)
  tryCatch({
    sc <- autoEstCont(sc)
    message("  -> Auto-estimated rho: ", sc$fit$rhoEst)
  }, error = function(e) {
    message("  -> autoEstCont failed, setting rho = 0.2")
    sc <- setContaminationFraction(sc, contFrac = 0.2)
  })

  adjusted_counts <- adjustCounts(sc, roundToInt = TRUE)
  sample_outdir <- file.path(output_dir, sample)
  if (dir.exists(sample_outdir)) {
  unlink(sample_outdir, recursive = TRUE)
}
  DropletUtils::write10xCounts(sample_outdir, adjusted_counts)
  message("  -> Output saved to: ", sample_outdir)
  rm(seu, sc, adjusted_counts)
  gc()
}

  obj[["percent_mitochondrion"]] <- PercentageFeatureSet(obj, features = mito_genes)
  obj[["percent_ribosome"]] <- PercentageFeatureSet(obj, pattern = "^RP[SL]")
  return(obj)
})

filter_sample <- function(obj, 
                          min_features = 200, 
                          max_features = 5000,
                          max_counts = 40000,
                          max_mt = 20) {
  subset(
    obj,
    subset = nFeature_RNA > min_features & 
             nFeature_RNA < max_features &
             nCount_RNA < max_counts &
             percent_mitochondrion < max_mt
  )
}

seob_list <- seob_list_filtered

for (i in seq_along(seob_list)) {
  s <- seob_list[[i]]
  nExp_poi <- round(0.08 * ncol(s))
  s <- FindNeighbors(s, dims = 1:30)
  s <- FindClusters(s, resolution = 0.4)

  homotypic.prop <- modelHomotypic(s$seurat_clusters)
  nExp_poi.adj <- round(nExp_poi * (1 - homotypic.prop))

  sweep.res.list <- paramSweep(s, PCs = 1:30, sct = FALSE)
  sweep.stats <- summarizeSweep(sweep.res.list, GT = FALSE)
  bcmvn <- find.pK(sweep.stats)

  pK <- as.numeric(as.character(bcmvn$pK[which.max(bcmvn$BCmetric)]))

  s <- doubletFinder(s, PCs = 1:30, pN = 0.25, pK = pK,
                     nExp = nExp_poi.adj, reuse.pANN = FALSE, sct = FALSE)

  seob_list[[i]] <- s
}
seob_list[[i]] <- RenameCells(seob_list[[i]], add.cell.id = sample_id)

seob_list_singlets <- lapply(seob_list, function(obj) {
  df_col <- grep("^DF.classifications_", colnames(obj@meta.data), value = TRUE)
  if (length(df_col) == 1) {
    singlet_cells <- colnames(obj)[which(obj@meta.data[[df_col]] == "Singlet")]
    subset(obj, cells = singlet_cells)
  } else {
    warning("")
    return(NULL)
  }
})
