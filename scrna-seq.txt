# ============================================================
# 2. Create Seurat objects
# ============================================================

seob_list_soupX <- list()

for (i in seq_along(sample_list)) {

  sample <- sample_list[i]
  time_point <- time_list[i]

  message(
    "Reading: ",
    sample,
    " (",
    time_point,
    ")"
  )

  data_path <- file.path(
    soupX_dir,
    sample
  )

  if (!dir.exists(data_path)) {
    stop(
      "Directory does not exist: ",
      data_path
    )
  }

  counts <- Read10X(
    data.dir = data_path
  )

  seob <- CreateSeuratObject(
    counts = counts,
    project = sample,
    min.cells = 3,
    min.features = 200
  )

  seob$orig.ident <- sample
  seob$time_info <- time_point

  seob_list_soupX[[sample]] <- seob
}

# 3. Calculate mitochondrial and ribosomal percentages
# ============================================================

mito_genes <- c(
  "ND1",
  "ND2",
  "ND3",
  "ND4",
  "ND4L",
  "ND5",
  "ND6",
  "COX1",
  "COX2",
  "COX3",
  "ATP8",
  "ATP6",
  "CYTB"
)

seob_list_soupX <- lapply(
  seob_list_soupX,
  function(obj) {

    obj[["percent_mitochondrion"]] <-
      PercentageFeatureSet(
        obj,
        features = mito_genes
      )

    obj[["percent_ribosome"]] <-
      PercentageFeatureSet(
        obj,
        pattern = "^RP[SL]"
      )

    return(obj)
  }
)


# ============================================================
# 4. QC filtering
# ============================================================

filter_sample <- function(
    obj,
    min_features = 200,
    max_features = 8000,
    max_counts = 40000,
    max_mt = 20
) {

  subset(
    obj,
    subset =
      nFeature_RNA > min_features &
      nFeature_RNA < max_features &
      nCount_RNA < max_counts &
      percent_mitochondrion < max_mt
  )
}

seob_list_filtered <- lapply(
  seob_list_soupX,
  filter_sample
)

# 6. QC violin plotting function

plot_qc_violin_no_points <- function(
    obj,
    features,
    output_file
) {

  p <- VlnPlot(
    obj,
    features = features,
    assay = "RNA",
    layer = "counts",
    pt.size = 0,
    raster = TRUE,
    ncol = length(features),
    combine = TRUE
  ) +
    NoLegend()

  ggsave(
    filename = output_file,
    plot = p,
    width = 22,
    height = 6,
    device = "pdf"
  )
}

# 9. DoubletFinder preprocessing
# ============================================================

set.seed(123)

seob_list <- seob_list_filtered

for (i in seq_along(seob_list)) {

  sample_id <- names(seob_list)[i]

  message(
    "\n========================================\n",
    "DoubletFinder preprocessing: ",
    sample_id,
    "\nCells: ",
    ncol(seob_list[[i]]),
    "\n========================================"
  )

  obj <- seob_list[[i]]

  obj <- NormalizeData(
    obj,
    verbose = FALSE
  )

  obj <- FindVariableFeatures(
    obj,
    selection.method = "vst",
    nfeatures = 2000,
    verbose = FALSE
  )

  obj <- ScaleData(
    obj,
    verbose = FALSE
  )

  obj <- RunPCA(
    obj,
    verbose = FALSE
  )

  seob_list[[i]] <- obj
}


# ============================================================
# 10. Run DoubletFinder separately for each sample
# ============================================================

for (i in seq_along(seob_list)) {

  sample_id <- names(seob_list)[i]

  s <- seob_list[[i]]

  message(
    "\n========================================\n",
    "Running DoubletFinder: ",
    sample_id,
    "\nCells before DoubletFinder: ",
    ncol(s),
    "\n========================================"
  )

  # ----------------------------------------------------------
  # Neighbor graph and clustering
  # ----------------------------------------------------------

  s <- FindNeighbors(
    s,
    dims = 1:30,
    verbose = FALSE
  )

  s <- FindClusters(
    s,
    resolution = 0.4,
    random.seed = 123,
    verbose = FALSE
  )


  # ----------------------------------------------------------
  # Expected doublet number
  # ----------------------------------------------------------

  nExp_poi <- round(
    0.08 * ncol(s)
  )


  # ----------------------------------------------------------
  # Homotypic doublet adjustment
  # ----------------------------------------------------------

  homotypic.prop <- modelHomotypic(
    s$seurat_clusters
  )

  nExp_poi.adj <- round(
    nExp_poi *
      (1 - homotypic.prop)
  )


  # ----------------------------------------------------------
  # Parameter sweep
  # ----------------------------------------------------------

  sweep.res.list <- paramSweep(
    s,
    PCs = 1:30,
    sct = FALSE
  )

  sweep.stats <- summarizeSweep(
    sweep.res.list,
    GT = FALSE
  )

  bcmvn <- find.pK(
    sweep.stats
  )


  # ----------------------------------------------------------
  # Remove NA BCmetric
  # ----------------------------------------------------------

  bcmvn <- bcmvn[
    !is.na(bcmvn$BCmetric),
    ,
    drop = FALSE
  ]

  if (nrow(bcmvn) == 0) {
    stop(
      "No valid pK value detected for sample: ",
      sample_id
    )
  }


  # ----------------------------------------------------------
  # Optimal pK
  # ----------------------------------------------------------

  pK <- as.numeric(
    as.character(
      bcmvn$pK[
        which.max(
          bcmvn$BCmetric
        )
      ]
    )
  )

  if (
    length(pK) != 1 ||
    is.na(pK)
  ) {
    stop(
      "Invalid pK for sample: ",
      sample_id
    )
  }


  message(
    "Sample: ",
    sample_id,
    "\nOptimal pK: ",
    pK,
    "\nExpected doublets before homotypic correction: ",
    nExp_poi,
    "\nExpected doublets after homotypic correction: ",
    nExp_poi.adj
  )


  # ----------------------------------------------------------
  # DoubletFinder
  # ----------------------------------------------------------

  s <- doubletFinder(
    s,
    PCs = 1:30,
    pN = 0.25,
    pK = pK,
    nExp = nExp_poi.adj,
    reuse.pANN = FALSE,
    sct = FALSE
  )

  seob_list[[i]] <- s
}


# ============================================================
# 11. Extract singlets
# ============================================================

seob_list_singlets <- lapply(
  seob_list,
  function(obj) {

    df_col <- grep(
      "^DF.classifications_",
      colnames(obj@meta.data),
      value = TRUE
    )

    if (length(df_col) != 1) {

      stop(
        "Expected exactly one DoubletFinder classification column, ",
        "but found ",
        length(df_col),
        "."
      )
    }

    singlet_cells <- colnames(obj)[
      obj@meta.data[[df_col]] ==
        "Singlet"
    ]

    subset(
      obj,
      cells = singlet_cells
    )
  }
)

stopifnot(
  length(seob_list_singlets) ==
    length(seob_list)
)

stopifnot(
  all(
    names(seob_list_singlets) ==
      names(seob_list)
  )
)
